[HarmonyPatch(typeof(StoreRequest), "ActivateRequest")]
internal static class VanillaRequestRefresh {           // 唯一的"闸门"
    private static bool Prefix(StoreRequest __0) {
        if (__0.id == "ul_rare_item") return false;      // 原版稀有品单：永不发
        string title = __0.title;
        if (!PrepaidOrders.HasActiveVanilla(title))
            return !PrepaidOrders.Load().BlocksTitle(title, PrepaidOrders.Day);
        return false;                                    // 自己已接同一标题 → 让路

[HarmonyPatch(typeof(StoreRequest), "FillRequest")]
internal static class VanillaCommissionCompletion {
    private static void Prefix(StoreRequest __instance, out bool __state)
        => __state = PlayerStore.Instance?.storeRequests.Contains(__instance) ?? false;
    private static void Postfix(StoreRequest __instance, bool __state) {
        if (__state && !PlayerStore.Instance.storeRequests.Contains(__instance)) {
            // 原版成交唯一的"前后可判"信号：进场时在列表里、出场时不在

            s.Save();
            PrepaidOrders.Load().RecordSettlement(__instance.title, PrepaidOrders.Day, completed: true);
            TaskOrders.CheckEmbargo(__instance.factionId);
        }
    }
}

### 1.4 踩坑与红线

1. **用 `title` 字符串当"与自身重复"的身份键**：`HasActiveVanilla(title)` 拿标题去比原版 `storeRequests`（`PrepaidOrders.cs:119-126`）。原版改文案就失配，会重复发单（推测）。Henchman 用 id 前缀是对的，抄这格要换 id。
2. **`ActivateRequest` 是全局静态方法**，Prefix 返回 false 会拦掉所有调用方（含原版 `StartDay` 内部）。白名单必须写全，否则一次手滑就让原版整个委托系统停摆。
3. **`Delay-2` 暗扣**（`PrepaidDefinition.cs:83`）：改表的人不知道有这个偏移，就会把"7 天"当"7 天"。
4. **一库两套存档口径**：`prepaid_schedule_v1` 是手写 pipe 串（`PrepaidSchedule.cs:45-48`），`tag_purchase_v1` 是 `JsonUtility` 整对象（`TagPurchaseState.cs:80-92`）。加字段要改两处不同风格的代码。
5. **`Decode` 用数组长度当版本号**（`8..13`，`PrepaidSchedule.cs:55-101`）：最省事、最脆的迁移法；少一段就静默丢一个子系统。
6. **`GenuineOnly` 判的是品类 `LUXURY_ITEM`**（`PrepaidOrders.cs:425`），不是 `realCondition`——名称骗人。按我们的红线，真伪只认 `ItemFeature.realCondition`，这格不能照抄。
7. 顾客注入每次都 `PrepaidOrders.Load()`（`CommissionVisitors.cs:30`），即每次生成顾客解析一遍整条 pipe 串；量大时该缓存 + 脏标记。

### 1.5 对我们的可用点

1. **"扩原版委托"= 三刀 + 一张自建表**：`FillRequest` Postfix 记账、`ActivateRequest` Prefix 拦重发、`HandleStoreRequests` Prefix 记到期，交付逻辑一行不改；新委托另开静态表，不要塞进 `storeRequests` 冒充原版。
2. **委托表的字段集直接可用**：`(id, 标题, 阵营, 物品, 数量, 期限, 定金, 尾款, 成功声望, 失败扣声望, 链/阶段/集章)`——比"按等级算奖励"好调，且天然支持"剧情链"（`chain+stage`）与"集章任务"（`stamps`）。
3. **`RequestItemIcon` 是"给原版卡画物品图标"的标准姿势**：临时 `DirectoryMaster.Create<GameItem>(id)` 取 sprite，`finally` 里递归 `Destroy`，`Dictionary` 缓存，`OnModItemDirectoryInit` 清空（`RequestItemIcon.cs:34-67`）。
4. **`SplitPaid` 最大余数法**（`TagPurchaseRules.cs:79-103`）：按报价比例把总额拆到每件、余数按小数部分降序补，`Σ == total`——批量结算拆账直接抄。
5. **`[HarmonyAfter("mod.faction_reputation_expansion")]` 是本库第一次见到的"和陌生模组抢同一块 UI"的显式声明**（`MarketRequestLayout.cs:69`）：只要你要重排别人也在重排的容器，就必须写。

### 2.2 验收判定：谁判、在哪判、判完怎么算

**它自己判，在原版议价平板上判。** 五步：

2. 玩家把货摆上柜台 → 打开原版议价平板 → `BuildGunOrder` **Prefix 返回 false**，整块面板换成自己的**逐件检测表**（`NegotiationTabletPatches.cs:95-282`，红/绿两色标每条读数）。
3. `DeliveryQuote.Inspect` 逐件：**硬条件全过**才算合格 → 加分档累加 → 按价格升序**贪心吃预算**（`DeliveryQuote.cs:50-79`）。
4. 玩家点"接受报价" → `OnOfferAcceptClick` **Prefix 拦截**（`NegotiationTabletPatches.cs:391-407`）→ `ConfirmDelivery`。
5. `ConfirmDelivery` 三重门 + **预览一致性闸门**：`MatchesPreview` 比较预算/需求版本/交期/逐件读数，任何一项变了就 `RefreshUI` + `Notify("柜台货品或验收结果有变化，请确认新结果。")` 并**不成交**（`ModEntry.cs:1918-1924`）。

### 3.2 是不是"派人出去→若干天后回来"：是，但主语反了

不是"玩家派人"，是"**玩家当赞助人**"：把「佣兵合作店家证书」放进**展示柜**（`Certificate.IsDisplayed()` 是唯一门牌，`Visits.cs:24`）→ 佣兵带悬赏单上门 → 玩家出装备（枪/弹/药/食物/烟闪）→ 佣兵走人 → N 夜后回来汇报；活着成功分你 `bounty × share%`（默认 50%），死了装备进 `lostGear`。**没有地点、没有场景、没有外出**——它是"异步任务 + 装备托管 + 单次结算"。

**派遣期间状态存哪**（`Merc.cs:18-32`）：`status` 五态（`Store.cs:11-19`：FREE 0 / MISSION 1 / RETURNING 2 / INJURED 3 / DEAD 4）、`outcome`（0 成功 / 1 惨胜）、`daysLeft`、`nextVisitDay`、`gearJson`（装备整包）、`contract`（含预先掷好的 `chance`/`deathPct`/`ammoShort`）。走人那一下（`Visits.cs:379-398`）：先 `Gear.Pack` 把柜台/补给箱上的货**塞进袋子 → 编码 → 销毁**，再 `mm.gearJson = json; mm.status = 1; mm.daysLeft = c.days;`。存档是**单键 `state`**（`Store.cs:95-98`，Newtonsoft 整对象）。

**怎么按天数推进**：`ModHook.OnGoingSleepLate += Nightly.Tick`（`MercDispatchMod.cs:40`）——每夜把 `status==1` 的 `daysLeft--`，到 0 就 `Resolve`；`status==3`（养伤）也 `daysLeft--`，到 0 复位成 FREE 并把 `nextVisitDay` 设为明天（`Nightly.cs:24-32`）。**单次 roll**：`Nightly.Resolve`（`Nightly.cs:52-71`）先掷 `chance`，没中再掷 `deathPct`，三个结果写进 `status/outcome`；回来时 `success = m.outcome == 0`（`Visits.cs:690`），**不重掷**——玩家在出发前看到的成功率就是最终判定。

**失败人会不会死：会，而且死了就是永久损失。** `DeathPct = 35 - 20×min(1, 药/需药) - 15×clamp01(药/需药-1)`，且 `contracts < 3` 时**恒为 0**（`Tuning.cs:505-513`，新手保护）。死了 status 4 → 下次开店`Messenger` 送信人上门 → 对话结束动作里：`Gear.Split(merc.gearJson)` 把整包拆成逐件全部塞进 `lostGear`（**只保留最近 6 件**，多的从最旧开始清）、`mercs.RemoveAll` 把人删掉、`respawnDay = 今天+3`、`AddPoints(-3)`（`Visits.cs:1005-1030`）。活着回来（含失败）走另一条：`AddCredit(pay)` → `Gear.ReturnToBox(gearJson)` 装备**回补给箱**，若之前有找回件则从 `lostGear` 移除（`Visits.cs:748-763`）。活着失败再落地为 `status=3; daysLeft=7`（养伤 7 天，期间 `Checkup` 上门）。成功 +`stars` 点、失败 +1 点 → `rank` → **槽位 1→4**（`Tuning.RANK_SLOTS`、`Store.cs:81-93`）。

**和原版"夜晚外出"的关系：没有关系。** 全库 `AfterhourLocation` / `MapUIManager` / `Travel` **0 命中**。它借的是"**过夜钩子**"（`OnGoingSleepLate`）推天数 + "**开店生成顾客**"三档（Early/Normal/Late）注入访客。也就是说：**它把"外出"抽象成了"时间流逝 + 回来的顾客"，一个场景都不用建。**

3. **多件验收 = 硬条件 + 加分档 + 预算贪心 + 预览一致闸门**，且**先记账后销毁**（② 的 `TryRecordPurchase` 是 RhodesPharma "先扣货后结账"的正面替代）。
4. **存盘三选一要说清**：pipe 串（省键、脆、靠字段数当版本）/ 逐字段键（最稳、最啰嗦，②）/ 单 JSON（最快、坏档全丢，③）。**凡是往原版集合里塞的对象，必须 Prefix 摘、Finalizer 挂回**（② `WriteSlotFile`），否则卸模即悬空。
5. **难度别写死数字**：三例各给了一种"随玩家变强而变难"的曲线——① 声望门槛、② 预算成长（质量档取 max + 销售额成长）、③ 星级 × 装备需求 + 带噪声的成功率。选与本模组经济耦合最紧的那条，别三条一起上。
